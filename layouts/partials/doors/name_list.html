{{/* Contributor names in a glossary term's credits. Each name is isolated, so Latin names keep their own order inside an Arabic sentence and the list runs in the page's direction. Input: dict with `names` and `sep` (", " or the Arabic "، "). */}} {{ $out := slice }} {{/* Each name is escaped before Markdown, as in term_title, so only the list's own markup is trusted. A name stays on one line; a long value (a list of links, say) may wrap. */}} {{/* Each name keeps the comma after it, so a line never starts with a comma, in Arabic as in English. */}} {{ $comma := strings.TrimRight " " .sep }} {{ $last := sub (len .names) 1 }} {{ range $i, $name := .names }}{{ $text := $name | htmlEscape | markdownify }}{{ $out = $out | append (printf "%s%s" (cond (lt (len (plainify $text)) 40) ` class="term-name"` "") $text (cond (lt $i $last) $comma "")) }}{{ end }} {{ return (delimit $out " " | safeHTML) }}
