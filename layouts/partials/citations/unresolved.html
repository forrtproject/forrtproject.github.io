{{/* Return diagnostics for recognisable author-year citations in a glossary definition. Like the browser matcher, resolve only unique first-author/year pairs in the term's own reference list. Explicit links already have a destination and need no warning. Input: dict "html" rendered definition "references" slice of APA strings. */}} {{- $references := dict -}} {{- range .references -}} {{- $meta := partialCached "citations/metadata" . . -}} {{- $pair := lower (printf "%s|%s" $meta.first $meta.year) -}} {{- $previous := index $references $pair | default (dict "count" 0) -}} {{- $references = merge $references (dict $pair (dict "count" (add $previous.count 1) "meta" $meta)) -}} {{- end -}} {{- $text := replaceRE `(?is)]*>.*?` "" .html | plainify | htmlUnescape -}} {{- $name := `(?:(?:[Vv]an|[Vv]on|[Dd]e|[Dd]en|[Dd]er|[Dd]el|[Ll]a|[Ll]e|[Dd]a|[Dd]i)\s+)*[\p{Lu}][\p{L}\p{M}'’.-]*(?:\s+[\p{Lu}][\p{L}\p{M}'’.-]*)*` -}} {{- $others := printf `(?:\s+et\s+al\.?|(?:,\s*%s)*(?:,?\s+(?:&|and)\s+%s))?` $name $name -}} {{- $date := `(?:[0-9]{4}[a-z]?|n\.d\.)` -}} {{- $pattern := printf `(%s)%s(?:(?:,\s*|\s+)(%s)|\s+\((%s)\))` $name $others $date $date -}} {{- $diagnostics := slice -}} {{- range findRE $pattern $text | uniq -}} {{- $first := replaceRE $pattern "$1" . -}} {{- $year := replaceRE $pattern "$2$3" . -}} {{- $pair := lower (printf "%s|%s" $first $year) -}} {{- $reference := index $references $pair | default (dict "count" 0) -}} {{- $reason := "" -}} {{- if eq $reference.count 0 -}} {{- $reason = "no matching listed reference" -}} {{- else if gt $reference.count 1 -}} {{- $reason = "ambiguous first author and year" -}} {{- else -}} {{- $label := replaceRE (printf `(?:(?:,\s*|\s+)%s|\s+\(%s\))$` $date $date) "" . | partial "citations/author-label" -}} {{- $short := partial "citations/author-label" $reference.meta.author -}} {{- $full := partial "citations/author-label" (delimit $reference.meta.names " ") -}} {{- if and (ne $label $short) (ne $label $full) -}}{{- $reason = "author list does not match the listed reference" -}}{{- end -}} {{- end -}} {{- if $reason -}}{{- $diagnostics = $diagnostics | append (dict "citation" . "reason" $reason) -}}{{- end -}} {{- end -}} {{- return $diagnostics -}}
