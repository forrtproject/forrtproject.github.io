{{/* Shortens a long author list in a full reference to its first three authors and "et al.", with a button that shows the rest in place (assets/js/doors/references.js). Takes a reference as rendered HTML. A reference with five authors or fewer, or whose authors aren't followed by a "(year" date, comes back unchanged. The hidden authors stay in the page, so copying gives the full APA text, and without JavaScript they show (layouts/partials/doors/assets.html). */}} {{- $html := trim (string .) " \n\r\t" -}} {{/* A surname (with "Jr." as the bibliography generator writes it) and initials. */}} {{- $name := `(?:[^,&…<>()]|&[a-z]+;|&#[0-9]+;)+?(?:,\s*[JS]r\.)?,\s*\p{Lu}\p{Ll}?\.(?:[\s-]*\p{Lu}\p{Ll}?\.)*\*?` -}} {{- $out := $html -}} {{- with findRESubmatch `^([^<(]*?)\s*\((?:[0-9]{4}|n\.d\.)` $html 1 -}} {{- $authors := index (index . 0) 1 -}} {{- $after := strings.TrimPrefix $authors $html -}} {{- $plain := htmlUnescape $authors -}} {{/* Bibliography entries can end in "& others" or "et al.": then, as after an ellipsis, the list is incomplete and the button doesn't give a number. */}} {{- $complete := not (findRE `…|\.\s?\.\s?\.|\bothers\b|\bet al\b` $plain) -}} {{- $marked := replaceRE `\bet al\.?|\bothers\b` "" (replaceRE $name "§" $plain) -}} {{/* Corporate authors such as "FORRT" are whatever is left between the separators. */}} {{- $count := add (len (findRE `§` $marked)) (len (findRE `[^\s,&…§.][^,&…§]*` $marked)) -}} {{- with and (gt $count 5) (findRE (printf `^(?:%s\s*,\s*(?:&(?:amp;)?\s*)?){2}%s` $name $name) $authors 1) -}} {{- $first := index . 0 -}} {{- $more := cond $complete (printf "Show all %d authors" $count) "Show more authors" -}} {{- $out = printf `%s%s, et al. %s` $first (strings.TrimPrefix $first $authors) $more $more $after -}} {{- end -}} {{- end -}} {{- $out | safeHTML -}}
