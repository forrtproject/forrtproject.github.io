{{/* Clusters map: a wheel with one segment per cluster, plus a legend, both linking to the clusters. Drawn from data/clusters_v4.json at build time, so it always matches the list. Expects dict: link_mode ("anchor" for #cluster-N on /clusters/, else /clusters/cluster-N/). */}} {{ with hugo.Data.clusters_v4 }}{{ with .clusters }} {{ $clusters := . }} {{ $palette := (partial "clusters/colors.html").arrowColors }} {{ $n := len $clusters }} {{ $cx := 200.0 }}{{ $cy := 200.0 }}{{ $R := 196.0 }}{{ $r := 104.0 }} {{ $gap := 1.2 }}{{/* degrees left blank between segments */}} {{ $step := div 360.0 $n }} {{ $items := slice }} {{ range $i, $c := $clusters }} {{ $num := int $c.number }} {{ $href := cond (eq $.link_mode "anchor") (printf "#cluster-%d" $num) ((printf "/clusters/cluster-%d/" $num) | relURL) }} {{ $a0 := math.ToRadians (add -90.0 (add (mul $i $step) (div $gap 2))) }} {{ $a1 := math.ToRadians (add -90.0 (sub (mul (add $i 1) $step) (div $gap 2))) }} {{ $mid := math.ToRadians (add -90.0 (mul (add $i 0.5) $step)) }} {{ $path := printf "M%.2f %.2fA%.0f %.0f 0 0 1 %.2f %.2fL%.2f %.2fA%.0f %.0f 0 0 0 %.2f %.2fZ" (add $cx (mul $R (math.Cos $a0))) (add $cy (mul $R (math.Sin $a0))) $R $R (add $cx (mul $R (math.Cos $a1))) (add $cy (mul $R (math.Sin $a1))) (add $cx (mul $r (math.Cos $a1))) (add $cy (mul $r (math.Sin $a1))) $r $r (add $cx (mul $r (math.Cos $a0))) (add $cy (mul $r (math.Sin $a0))) }} {{ $lr := div (add $R $r) 2 }} {{ $items = $items | append (dict "num" $num "name" $c.name "href" $href "path" $path "color" (index $palette (mod $i (len $palette))) "x" (add $cx (mul $lr (math.Cos $mid))) "y" (add $cy (mul $lr (math.Sin $mid))) "subs" (len $c.sub_clusters)) }} {{ end }} {{ end }}{{ end }}
