{{- /* Publication titles from Crossref: plain text plus , and markup, sometimes with HTML entities. Escape everything, then restore only those tags (scp as small caps). static/js/featured-resources.js has the same rule (richText). */ -}} {{- $s := . | default "" | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace | htmlEscape -}} {{- $s = replaceRE `<i>(.*?)</i>` "$1" $s -}} {{- $s = replaceRE `<em>(.*?)</em>` "$1" $s -}} {{- $s = replaceRE `<scp>(.*?)</scp>` `$1` $s -}} {{- return ($s | safeHTML) -}}
